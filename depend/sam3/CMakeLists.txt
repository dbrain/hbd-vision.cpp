# sam3.cpp — SAM 3 promptable concept segmentation.
#
# VENDORED, not a submodule. Upstream is PABannier/sam3.cpp @
# 01832ef85fcc8eb6488f1d01cd247f07e96ff5a9 (v1.0.0), MIT — see LICENSE. What we
# actually compile is two files, and the upstream repo carries 22 MB of nested
# ggml we never initialize plus 12 MB of media; carrying a fork of that to hold
# a few hundred lines of CUDA support was not worth it.
#
# Local changes on top of upstream (all in sam3.cpp, all marked in-source):
#   - CUDA backend init + SAM3_CUDA_DEVICE
#   - sam3_win_part / sam3_win_unpart / sam3_mean_rows / sam3_attn_ext, the
#     rewrites of ops the CUDA backend has no kernel for, behind SAM3_LEGACY_OPS
# Re-syncing with upstream means diffing sam3.cpp against that rev.
#
# It builds against THIS tree's ggml, not upstream's pinned one.
#
# stb: the upstream single translation unit compiles the stb image/write
# implementations into itself, and vision.cpp already ships those symbols in the
# `stb` target. Force-including the real headers does NOT stop that — stb
# deliberately puts its implementation block OUTSIDE the include guard
# (`stb_image.h`: the guard closes at `#endif // STBI_INCLUDE_STB_IMAGE_H`, and
# `#ifdef STB_IMAGE_IMPLEMENTATION` opens two lines later), so on sam3.cpp's own
# `#define STB_IMAGE_IMPLEMENTATION` + re-include the declarations are suppressed
# and the implementation lands anyway: a duplicate-symbol link failure in any
# static build. Interpose a guarded shim on the include path instead. The
# force-include pulls the real declarations in through the shim, and sam3.cpp's
# re-include hits the shim's own guard and expands to nothing.

# The real headers come from the `stb` target's interface, not a local path.
get_target_property(SAM3_STB_INCLUDE stb INTERFACE_INCLUDE_DIRECTORIES)
list(GET SAM3_STB_INCLUDE 0 SAM3_STB_INCLUDE)

set(SAM3_STB_SHIM ${CMAKE_CURRENT_BINARY_DIR}/stb-shim)
foreach(hdr stb_image stb_image_write)
  file(
    WRITE ${SAM3_STB_SHIM}/${hdr}.h
    "// Generated by depend/sam3/CMakeLists.txt — see the comment there.\n"
    "#ifndef SAM3_STB_SHIM_${hdr}\n"
    "#define SAM3_STB_SHIM_${hdr}\n"
    "#include \"${SAM3_STB_INCLUDE}/${hdr}.h\"\n"
    "#endif\n")
endforeach()

add_library(sam3 STATIC ${CMAKE_CURRENT_SOURCE_DIR}/sam3.cpp)
# BEFORE: the shim has to win over the real stb include dir that `stb` exports.
target_include_directories(sam3 BEFORE PRIVATE ${SAM3_STB_SHIM})
target_include_directories(sam3 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(sam3 PUBLIC ggml stb)
target_compile_features(sam3 PUBLIC cxx_std_17)
target_compile_options(sam3 PRIVATE ${VISP_NO_WARNINGS})
set_target_properties(sam3 PROPERTIES POSITION_INDEPENDENT_CODE ON)

# sam3.cpp uses INT_MAX but never includes <climits>; it only ever compiled
# because stb's implementation block pulls <limits.h> in behind it. Now that the
# shim suppresses that block, supply the header sam3.cpp actually depends on.
if(MSVC)
  target_compile_options(sam3 PRIVATE /FIclimits /FIstb_image.h /FIstb_image_write.h)
else()
  # SHELL: keeps each flag/argument pair together — CMake otherwise de-duplicates
  # the two bare `-include` tokens and leaves the headers as input files.
  target_compile_options(
    sam3 PRIVATE "SHELL:-include climits" "SHELL:-include stb_image.h"
                 "SHELL:-include stb_image_write.h")
endif()

# Quantizes a converted .ggml to q4_0/q4_1/q8_0. Upstream ships this as an
# example; it is the only supported way to produce the q8_0 model the service
# runs, so it is a first-class target here rather than something to hand-compile.
add_executable(sam3-quantize ${CMAKE_CURRENT_SOURCE_DIR}/tools/quantize.cpp)
target_link_libraries(sam3-quantize PRIVATE ggml)
target_compile_options(sam3-quantize PRIVATE ${VISP_NO_WARNINGS})

# Single-process PCS driver. vision-server forks its CUDA worker, which makes it
# useless as an nsys/ncu target — this runs the same calls in one process.
add_executable(sam3-bench ${CMAKE_CURRENT_SOURCE_DIR}/tools/bench.cpp)
target_link_libraries(sam3-bench PRIVATE sam3 ggml stb)
target_compile_definitions(sam3-bench PRIVATE ${VISP_DEFINITIONS})
target_compile_options(sam3-bench PRIVATE ${VISP_NO_WARNINGS})
